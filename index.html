def contents():
    return {
        "About Me": "1-3",
        "LIFE": "4-11",
        "Rewind": "11-18",
        "Has Been": "19-26",
        "portfolio": "27-34",
        "SECOND Coming": "35-43",
    }

def select_chapter(name):
    name = " ".join(name.strip().split()).lower()
    for title, pages in contents().items():
        if title.lower() == name:
            return title, pages
    return "Chapter not found"

def main():
    name = input("")▌
    result = select_chapter(name)
    print(result)

main()
